### B. Structure visualization and organization

- A bounded, readable map built from actual repository folders/files and structural containment.

- Portable export of the implemented structural view, using the PRD’s Archify architecture.

Do not claim that a containment map is a dependency graph. Default groups must come from observed structure or explicit user organization, not AI-generated architectural guesses.

“Organize” means organize the VIEW. Never move, rename, delete, or rewrite imported source files.

def _detect_language(path: Path) -> tuple[str, bool]:
    """Return (language, is_binary)."""
    suffix = path.suffix.lower()
    name = path.name.lower()
    if suffix in _BINARY_EXTENSIONS:
        return ("binary", True)
    lang = _LANGUAGE_MAP.get(suffix) or _LANGUAGE_MAP.get(name, "unknown")
    return (lang, False)


def _sha256_file(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(65536), b""):
            h.update(chunk)
    return h.hexdigest()


def _sha256_str(s: str) -> str:
    return hashlib.sha256(s.encode()).hexdigest()


def _build_inventory(
    snapshot_id: str,
    project_dir: Path,
    ignored_dirs: set[str] | None = None,
) -> list[FileRecord]:
    if ignored_dirs is None:
        ignored_dirs = {".git", "node_modules", "__pycache__", ".venv", "dist", "build"}

    records: list[FileRecord] = []
    resolved_root = project_dir.resolve()

    for current_dir, dirs, filenames in os.walk(project_dir, followlinks=False):
        current_path = Path(current_dir)
        dirs[:] = sorted(
            d for d in dirs
            if d.casefold() not in ignored_dirs
            and not (current_path / d).is_symlink()
        )
        for filename in sorted(filenames):
            src = current_path / filename
            if src.is_symlink():
                continue
            try:
                rel = src.resolve().relative_to(resolved_root).as_posix()
                size = src.stat().st_size
            except (OSError, ValueError):
                continue

            lang, is_binary = _detect_language(src)
            content_hash = _sha256_file(src)
            file_id = _sha256_str(f"{snapshot_id}:{rel}")[:32]

            records.append(FileRecord(
                id=file_id,
                snapshot_id=snapshot_id,
                path=rel,
                name=src.name,
                size=size,
                content_hash=content_hash,
                is_text=not is_binary,
                language=lang,
                language_basis="extension",
            ))

    return records

def get_inventory(snapshot_id: str, cursor: str | None = None, limit: int = 500) -> FilePage:

        total=total,
        cursor=next_cursor,

def build_capability_report(snapshot_id: str) -> CapabilityReport:
    """Build a capability report from the snapshot inventory."""
    page = get_inventory(snapshot_id, limit=100000)
    files = []
    parsed = text_only = binary = excluded = 0

    for rec in page.files:
        if rec.excluded:
            excluded += 1
            level = "excluded"
            syntax = ref = summary = False
        elif not rec.is_text:
            binary += 1
            level = "binary"
            syntax = ref = summary = False
        elif rec.language in ("python",):
            parsed += 1
            level = "full"
            syntax = True
            ref = False  # import names extracted but not fully resolved
            summary = True
        else:
            text_only += 1
            level = "text_only"
            syntax = False
            ref = False
            summary = True

        caps = FileCapability(
            file_id=rec.id,
            path=rec.path,
            language=rec.language,
            syntax_extraction=syntax,
            reference_resolution=ref,
            summary_eligible=summary,
            level=level,
            limitations=[] if level != "text_only" else [
                "No syntax parser available for this language; text search only."
            ],
        )
        files.append(caps)

    return CapabilityReport(
        snapshot_id=snapshot_id,
        files=files,
        parsed_count=parsed,
        text_only_count=text_only,
        binary_count=binary,
        excluded_count=excluded,
        limitations=["Deep semantic analysis is available for Python only in this build."],
    )

def read_source_lines(
    snapshot_id: str,
    file_id: str,
    line_start: int = 1,
    line_end: int | None = None,
    max_lines: int = 500,
) -> "SourceSlice":
    """Read bounded source lines from an immutable snapshot file."""
    from app.models.v1.source import SourceSlice

    rec = get_file_record(snapshot_id, file_id)

    # Locate the actual file in the project directory
    # For v1 ZIP imports the files live in the legacy projects dir under the
    # project_id.  For GitHub they live in a dedicated extraction directory.
    snap = get_snapshot(snapshot_id)

    if snap.source.kind == SourceKind.ZIP:
        # Find the project directory by looking up the legacy project record
        projects_root = _projects_root()
        # Try to find matching project_id in v1 project records
        proj_path = _v1_root() / f"project_{snap.project_id}.json"
        project_dir: Path | None = None
        if proj_path.exists():
            try:
                from app.models.v1.snapshot import Project as V1Project
                proj = V1Project.model_validate(_read_json(proj_path))
                # The files live in the legacy projects root
                candidate = projects_root / snap.project_id
                if candidate.is_dir():
                    project_dir = candidate
            except Exception:
                pass
        if project_dir is None:
            # Fall back: scan legacy projects root
            for d in projects_root.iterdir():
                if d.is_dir() and not d.name.startswith("."):
                    project_dir = d
                    break

        if project_dir is None:
            return _error_slice(snapshot_id, file_id, rec.path, "Project directory not found.")

        file_path = project_dir / rec.path
    else:
        # GitHub: files live in a snapshot-specific extraction dir
        snap_dir = _v1_root() / snapshot_id / "files"
        file_path = snap_dir / rec.path

    if not file_path.exists() or file_path.is_symlink():
        return _error_slice(snapshot_id, file_id, rec.path, "Source file not available.")

    try:
        text = file_path.read_text(encoding="utf-8", errors="replace")
    except OSError:
        return _error_slice(snapshot_id, file_id, rec.path, "Could not read source file.")

    lines = text.splitlines()
    total_lines = len(lines)
    if line_start < 1:
        line_start = 1
    if line_end is None or line_end > total_lines:
        line_end = total_lines
    if line_end - line_start + 1 > max_lines:
        line_end = line_start + max_lines - 1
        truncated = True
    else:
        truncated = False

    selected = lines[line_start - 1 : line_end]
    content = "\n".join(selected)

    from app.models.v1.source import SourceSlice
    return SourceSlice(
        file_id=file_id,
        snapshot_id=snapshot_id,
        path=rec.path,
        line_start=line_start,
        line_end=min(line_end, total_lines),
        total_lines=total_lines,
        content=content,
        truncated=truncated,
        content_hash=rec.content_hash,
    )

def _build_from_records(snapshot_id: str, page: FilePage, max_entities: int) -> Graph:

    # 1. Collect directory paths
    # ------------------------------------------------------------------
    dir_paths: set[str] = set()
    for rec in page.files:
        parts = rec.path.split("/")
        for depth in range(1, len(parts)):
            dir_paths.add("/".join(parts[:depth]))

    # 2. Build entities
    # ------------------------------------------------------------------

    entity_index: dict[str, GraphEntity] = {}

    # Root repo entity
    root_id = _stable_id(snapshot_id, ".")
    root_entity = GraphEntity(
        id=root_id,
        kind=EntityKind.REPOSITORY,
        label="(repository root)",
        path=".",
        child_count=0,
    )
    entities.append(root_entity)
    entity_index["."] = root_entity

    # Directory entities
    for dir_path in sorted(dir_paths):
        eid = _stable_id(snapshot_id, dir_path)
        name = dir_path.rsplit("/", 1)[-1]
        ent = GraphEntity(
            id=eid,
            kind=EntityKind.DIRECTORY,
            label=name,
            path=dir_path,
            parent_id=None,  # set below
            child_count=0,
        )
        entities.append(ent)
        entity_index[dir_path] = ent

    # Set parent IDs for directories
    for dir_path, ent in entity_index.items():
        if dir_path == ".":
            continue
        if "/" in dir_path:
            parent_path = dir_path.rsplit("/", 1)[0]
        else:
            parent_path = "."
        parent = entity_index.get(parent_path)
        if parent:
            ent.parent_id = parent.id
            parent.child_count += 1

    # File entities
    for rec in page.files:
        parts = rec.path.split("/")
        parent_path = "/".join(parts[:-1]) if len(parts) > 1 else "."
        parent = entity_index.get(parent_path)
        parent_id = parent.id if parent else root_id
        if parent:
            parent.child_count += 1

        ent = GraphEntity(
            id=eid,
            kind=EntityKind.FILE,
            label=rec.name,
            path=rec.path,
            file_id=rec.id,
            parent_id=parent_id,
            metadata={
                "language": rec.language,
                "size": rec.size,
                "is_text": rec.is_text,
            },
        )
        entities.append(ent)
        entity_index[rec.path] = ent

    # 3. Truncate if needed
    # ------------------------------------------------------------------

    # 4. Build contains relations
    # ------------------------------------------------------------------
    relations: list[GraphRelation] = []
    entity_ids = {e.id for e in entities}

                evidence_ids=[],

    # ------------------------------------------------------------------

            "Dependency/import edges are provided by the Dependencies integration slot."
        ],
        truncated=truncated,
        total_entities=total_entities,
        returned_entities=len(entities),
    )

        snapshot_id=snapshot_id,
        entities=entities,

from pathlib import Path

ALLOWED_HOST = "github.com"
MAX_ARCHIVE_BYTES = 50 * 1024 * 1024  # 50 MiB – same as ZIP limit
GITHUB_API_BASE = "https://api.github.com"


class GitHubImportError(ValueError):
    pass


class GitHubImportTooLargeError(GitHubImportError):
    pass


_OWNER_RE = re.compile(r"^[A-Za-z0-9](?:[A-Za-z0-9\-]{0,37}[A-Za-z0-9])?$")
_REPO_RE = re.compile(r"^[A-Za-z0-9_.\-]{1,100}$")
_REF_RE = re.compile(r"^[A-Za-z0-9_./\-]{1,200}$")
_COMMIT_RE = re.compile(r"^[a-f0-9]{40}$")


def _safe_urlopen(url: str, timeout: int = 30) -> bytes:
    """Fetch URL with basic safety checks (no arbitrary redirect, no credentials)."""
    parsed = urllib.parse.urlparse(url)
    if parsed.scheme != "https" or parsed.netloc not in (ALLOWED_HOST, "api.github.com", "codeload.github.com"):
        raise GitHubImportError(f"URL host is not allowlisted: {parsed.netloc}")
    if parsed.username or parsed.password:
        raise GitHubImportError("Credentials must not be embedded in URLs.")

    req = urllib.request.Request(url, headers={
        "Accept": "application/vnd.github.v3+json",
        "User-Agent": "Grepo/1.0 read-only-importer",
    })
    with urllib.request.urlopen(req, timeout=timeout) as response:  # noqa: S310
        # Only follow redirects to allowlisted hosts (urllib follows by default)
        final_url = response.geturl()
        final_host = urllib.parse.urlparse(final_url).netloc
        if final_host not in (ALLOWED_HOST, "api.github.com", "codeload.github.com",
                               "objects.githubusercontent.com"):
            raise GitHubImportError(f"Redirect to non-allowlisted host: {final_host}")
        return response.read()


def parse_github_url(url: str) -> tuple[str, str]:
    """Parse owner and repo from a public GitHub HTTPS URL.

    Returns (owner, repo).  Raises GitHubImportError for invalid/unsafe URLs.
    """
    url = url.strip()
    parsed = urllib.parse.urlparse(url)
    if parsed.scheme not in ("https", ""):
        raise GitHubImportError("Only HTTPS GitHub URLs are supported.")
    if parsed.netloc and parsed.netloc != ALLOWED_HOST:
        raise GitHubImportError(f"Only github.com URLs are supported, got: {parsed.netloc}")
    if parsed.username or parsed.password:
        raise GitHubImportError("Credentials must not be embedded in the URL.")

    path = parsed.path.rstrip("/")
    parts = [p for p in path.split("/") if p]
    if len(parts) < 2:
        raise GitHubImportError("URL must specify owner/repository.")

    owner, repo = parts[0], parts[1]
    if repo.endswith(".git"):
        repo = repo[:-4]
    if not _OWNER_RE.fullmatch(owner):
        raise GitHubImportError("Invalid repository owner name.")
    if not _REPO_RE.fullmatch(repo):
        raise GitHubImportError("Invalid repository name.")

    return owner, repo


def resolve_ref_to_commit(owner: str, repo: str, ref: str = "HEAD") -> tuple[str, str]:
    """Resolve a branch/tag/HEAD to an immutable commit SHA.

    Returns (resolved_commit_sha, display_ref).
    Raises GitHubImportError if the repository is not found or not public.
    """
    import json as _json

    if not _REF_RE.fullmatch(ref):
        raise GitHubImportError("Invalid ref format.")

    # Check repo visibility first
    repo_url = f"{GITHUB_API_BASE}/repos/{owner}/{repo}"
    try:
        repo_data = _json.loads(_safe_urlopen(repo_url))
    except (GitHubImportError, Exception) as e:
        raise GitHubImportError(f"Could not access GitHub repository: {e}") from e

    if repo_data.get("private", True):
        raise GitHubImportError(
            "This repository is private. Only public repositories are supported."
        )

    # Resolve the ref
    if ref == "HEAD" or ref == "":
        ref = repo_data.get("default_branch", "main")

    commit_url = f"{GITHUB_API_BASE}/repos/{owner}/{repo}/commits/{ref}"
    try:
        commit_data = _json.loads(_safe_urlopen(commit_url))
    except Exception as e:
        raise GitHubImportError(f"Could not resolve ref '{ref}': {e}") from e

    sha = commit_data.get("sha", "")
    if not _COMMIT_RE.fullmatch(sha):
        raise GitHubImportError("GitHub API returned an invalid commit SHA.")

    return sha, ref


def download_archive(owner: str, repo: str, commit: str, dest_dir: Path) -> Path:

    Returns path to the downloaded archive file.
    Uses the pinned commit, not a moving branch reference.

        raise GitHubImportError("Invalid commit SHA.")

    archive_url = f"https://codeload.github.com/{owner}/{repo}/zip/{commit}"
    req = urllib.request.Request(archive_url, headers={
        "User-Agent": "Grepo/1.0 read-only-importer",
    })

    archive_path = dest_dir / "github_archive.zip"
    downloaded = 0

    try:
        with urllib.request.urlopen(req, timeout=120) as response:  # noqa: S310
            final_host = urllib.parse.urlparse(response.geturl()).netloc

                                  "objects.githubusercontent.com"):
                raise GitHubImportError(f"Redirect to non-allowlisted host: {final_host}")
            with archive_path.open("wb") as out:
                while chunk := response.read(65536):
                    downloaded += len(chunk)
                    if downloaded > MAX_ARCHIVE_BYTES:
                        raise GitHubImportTooLargeError(

    url: str,
    ref: str = "HEAD",
) -> tuple[str, str, str, str, str, Path]:
    """Top-level GitHub import workflow.

    Returns (owner, repo, resolved_commit, display_ref, project_name, files_dir).

    The files_dir directory is owned by the caller and must be cleaned up.
    """
    from fastapi import UploadFile
    from app.services.project_archive import extract_project_archive

    owner, repo = parse_github_url(url)
    resolved_commit, display_ref = resolve_ref_to_commit(owner, repo, ref)

    staging = Path(tempfile.mkdtemp(prefix="grepo_gh_"))
    files_dir = staging / "files"
    archive_path = staging / "github_archive.zip"
    project_name = f"{owner}/{repo}"

    try:
        download_archive(owner, repo, resolved_commit, staging)

        # Use the existing safe archive extraction
        class _FakeUpload:
            filename = f"{repo}.zip"
            size = archive_path.stat().st_size

            async def read(self, n: int = -1) -> bytes:
                # Streaming read for the extraction service
                if not hasattr(self, "_f"):
                    self._f = open(archive_path, "rb")  # noqa: SIM115
                return self._f.read(n)

        fake_upload = _FakeUpload()

        import zipfile

        # Strip the top-level directory that GitHub adds to its archives
        with zipfile.ZipFile(archive_path) as zf:
            top_dirs = {info.filename.split("/")[0] for info in zf.infolist() if "/" in info.filename}

        files_dir.mkdir(exist_ok=True)

        # Re-extract without the top-level prefix using the safe extractor
        from app.services.project_archive import (
            IGNORED_DIRECTORIES,
            _validated_entries,
            _extract_entries,
        )

        with zipfile.ZipFile(archive_path) as archive:
            entries = _validated_entries(archive)
            # Strip the common prefix (GitHub adds owner-repo-commit/)
            prefix = ""
            if top_dirs and len(top_dirs) == 1:
                prefix = top_dirs.pop() + "/"

            if prefix:
                for entry in entries:
                    p = entry.info.filename
                    if p.startswith(prefix):
                        entry.info.filename = p[len(prefix):]

            _extract_entries(archive, entries, files_dir)

    except Exception:
        shutil.rmtree(staging, ignore_errors=True)
        raise

    archive_path.unlink(missing_ok=True)
    return owner, repo, resolved_commit, display_ref, project_name, files_dir

// ── Error handling ─────────────────────────────────────────────────────────

export class ApiError extends Error {
  constructor(
    public readonly status: number,
    message: string,
    public readonly code?: string,
  ) {
    super(message)
    this.name = 'ApiError'
  }
}

async function handleResponse<T>(res: Response): Promise<T> {
  if (res.ok) {
    return res.json() as Promise<T>
  }
  let message = `API returned ${res.status}`
  let code: string | undefined
  try {
    const payload: unknown = await res.json()
    if (typeof payload === 'object' && payload !== null) {
      const p = payload as Record<string, unknown>
      if ('detail' in p && typeof p.detail === 'string') message = p.detail
      if ('error' in p && typeof p.error === 'object' && p.error !== null) {
        const e = p.error as Record<string, unknown>
        if (typeof e.message === 'string') message = e.message
        if (typeof e.code === 'string') code = e.code
      }
    }
  } catch {
    // ignore parse errors
  }
  throw new ApiError(res.status, message, code)
}

  snapshotId: string,
  cursor?: string,

  if (cursor) params.set('cursor', cursor)
  const res = await fetch(

/** Fetch all file records across pages */
export async function getAllFiles(projectId: string, snapshotId: string): Promise<FilePage['files']> {

  let cursor: string | undefined
  do {
    const page = await getInventory(projectId, snapshotId, cursor)
    all.push(...page.files)
    cursor = page.cursor ?? undefined
  } while (cursor)
  return all
}

interface TreeNode {
  id: string
  name: string
  path: string
  kind: 'directory' | 'file'
  language?: string
  fileId?: string
  size?: number
  children: TreeNode[]
  depth: number
}

function buildTree(files: FileRecord[]): TreeNode[] {
  const root: Map<string, TreeNode> = new Map()
  const allNodes: Map<string, TreeNode> = new Map()

  // Build directory nodes first
  const dirPaths = new Set<string>()
  for (const f of files) {
    const parts = f.path.split('/')
    for (let i = 1; i < parts.length; i++) {
      dirPaths.add(parts.slice(0, i).join('/'))
    }
  }

  for (const dirPath of Array.from(dirPaths).sort()) {
    const parts = dirPath.split('/')
    const node: TreeNode = {
      id: `dir:${dirPath}`,
      name: parts[parts.length - 1],
      path: dirPath,
      kind: 'directory',
      children: [],
      depth: parts.length - 1,
    }
    allNodes.set(dirPath, node)
  }

  // Build file nodes
  for (const f of files) {
    const node: TreeNode = {
      id: `file:${f.path}`,
      name: f.name,
      path: f.path,
      kind: 'file',
      language: f.language !== 'unknown' ? f.language : undefined,
      fileId: f.id,
      size: f.size,
      children: [],
      depth: f.path.split('/').length - 1,
    }
    allNodes.set(f.path, node)
  }

  // Build parent-child relationships
  for (const [path, node] of allNodes) {
    const parts = path.split('/')
    if (parts.length === 1) {
      root.set(path, node)
    } else {
      const parentPath = parts.slice(0, -1).join('/')
      const parent = allNodes.get(parentPath)
      if (parent) {
        parent.children.push(node)
      } else {
        root.set(path, node)
      }
    }
  }

  // Sort: directories before files, then alphabetically
  function sortNodes(nodes: TreeNode[]): TreeNode[] {
    nodes.sort((a, b) => {
      if (a.kind !== b.kind) return a.kind === 'directory' ? -1 : 1
      return a.name.localeCompare(b.name)
    })
    for (const n of nodes) sortNodes(n.children)
    return nodes
  }

  return sortNodes(Array.from(root.values()))
}

function filterTree(nodes: TreeNode[], query: string): TreeNode[] {
  if (!query) return nodes
  const q = query.toLowerCase()

  function matchNode(node: TreeNode): TreeNode | null {
    if (node.kind === 'file') {
      return node.path.toLowerCase().includes(q) ? node : null
    }
    const matchedChildren = node.children.flatMap(c => {
      const m = matchNode(c)
      return m ? [m] : []
    })
    if (matchedChildren.length > 0 || node.path.toLowerCase().includes(q)) {
      return { ...node, children: matchedChildren }
    }
    return null
  }

  return nodes.flatMap(n => {
    const m = matchNode(n)
    return m ? [m] : []
  })
}

function TreeNodeRow({
  node,
  selectedPath,
  expandedPaths,
  onSelect,
  onToggle,
  depth,

  const isExpanded = expandedPaths.has(node.path)
  const isSelected = selectedPath === node.path
  const isDir = node.kind === 'directory'
  const langShort = node.language ? (LANG_SHORT[node.language] ?? node.language.slice(0, 4)) : undefined

    if (isDir) {
      onToggle(node.path)
    } else {
      onSelect(node)
    }

    if (e.key === 'Enter' || e.key === ' ') {
      e.preventDefault()
      handleClick()
    }
    if (e.key === 'ArrowRight' && isDir && !isExpanded) {
      onToggle(node.path)
    }
    if (e.key === 'ArrowLeft' && isDir && isExpanded) {
      onToggle(node.path)
    }
  }

  return (
    <>
      <button
        type="button"
        className={`tree-node ${isSelected ? 'tree-node--selected' : ''}`}
        style={{ paddingLeft: `${8 + depth * 14}px` }}
        onClick={handleClick}
        onKeyDown={handleKeyDown}
        aria-expanded={isDir ? isExpanded : undefined}
        aria-selected={isSelected}
        title={node.path}
        tabIndex={0}
      >
        {isDir ? (
          <ChevronIcon open={isExpanded} />

          key={child.id}
          node={child}
          selectedPath={selectedPath}
          expandedPaths={expandedPaths}
          onSelect={onSelect}
          onToggle={onToggle}
          depth={depth + 1}
        />
      ))}
    </>
  )
}

      else next.add(path)

  }, [])

  const handleSelect = useCallback((node: TreeNode) => {
    if (node.kind === 'file') {
      const file = files.find(f => f.path === node.path)

    }

        <input
          ref={searchRef}

        className="tree-panel__scroll"

// ── Layout engine (simple tidy tree layout) ───────────────────────────────

interface LayoutNode {
  entity: GraphEntity
  x: number
  y: number
  width: number
  height: number
  children: LayoutNode[]
  depth: number
}

const NODE_W = 140
const NODE_H = 36
const H_GAP = 20
const V_GAP = 20
const PADDING = 40

function buildLayoutTree(graph: Graph): LayoutNode[] {
  const entityMap = new Map(graph.entities.map(e => [e.id, e]))
  const childrenMap = new Map<string, string[]>()

  for (const rel of graph.relations) {
    if (rel.kind === 'contains') {
      const list = childrenMap.get(rel.source_id) ?? []
      list.push(rel.target_id)
      childrenMap.set(rel.source_id, list)
    }
  }

  const roots = graph.entities.filter(e => !e.parent_id || e.kind === 'repository')

  let counter = { x: PADDING }

  function buildNode(entity: GraphEntity, depth: number): LayoutNode {
    const childIds = childrenMap.get(entity.id) ?? []
    const children = childIds
      .map(id => entityMap.get(id))
      .filter((e): e is GraphEntity => !!e)
      .map(e => buildNode(e, depth + 1))

    const node: LayoutNode = {
      entity,
      x: 0,
      y: PADDING + depth * (NODE_H + V_GAP),
      width: NODE_W,
      height: NODE_H,
      children,
      depth,
    }

    if (children.length === 0) {
      node.x = counter.x
      counter.x += NODE_W + H_GAP
    } else {
      const first = children[0]
      const last = children[children.length - 1]
      node.x = (first.x + last.x + last.width) / 2 - NODE_W / 2
    }

    return node
  }

  return roots.map(r => buildNode(r, 0))
}

function flattenLayout(nodes: LayoutNode[]): LayoutNode[] {
  const result: LayoutNode[] = []
  function visit(n: LayoutNode) {
    result.push(n)
    n.children.forEach(visit)
  }
  nodes.forEach(visit)
  return result
}

export function generateStructuralHtml(
  snapshot: Snapshot,
  graph: Graph,
  manifest: ExportManifest,

  const title = escHtml(
    snapshot.source.repo
      ? `${snapshot.source.owner ?? ''}/${snapshot.source.repo}`
      : `Snapshot ${snapshot.id.slice(0, 8)}`,

  const NODE_W = 140
  const NODE_H = 36
  const H_GAP = 20
  const V_GAP = 20
  const PADDING = 40

  const entityMap = new Map(graph.entities.map(e => [e.id, e]))
  const childrenMap = new Map<string, string[]>()
  for (const rel of graph.relations) {

/**
 * Trigger a browser download of the structural HTML export.
 */
export function downloadStructuralExport(
  snapshot: Snapshot,
  graph: Graph,
  manifest: ExportManifest,
): void {
  const html = generateStructuralHtml(snapshot, graph, manifest)
  const blob = new Blob([html], { type: 'text/html;charset=utf-8' })
  const url = URL.createObjectURL(blob)
  const a = document.createElement('a')
  const name = snapshot.source.repo ?? `snapshot-${snapshot.id.slice(0, 8)}`
  a.href = url
  a.download = `grepo-${name.replace(/[^a-z0-9\-_.]/gi, '_')}-structure.html`
  document.body.appendChild(a)
  a.click()
  document.body.removeChild(a)
  URL.revokeObjectURL(url)
}

import { useEffect, useState } from 'react'
import { useNavigate, useParams } from 'react-router-dom'
import { IntegrationPlaceholder } from '../../components/slots/IntegrationPlaceholder'
import { useWorkspace } from '../../contexts/WorkspaceContext'
import { getCapabilities } from '../../services/v1/api'
import type { CapabilityReport } from '../../types/v1'

function formatBytes(n: number): string {
  if (n < 1024) return `${n} B`
  if (n < 1024 * 1024) return `${(n / 1024).toFixed(1)} KB`
  return `${(n / (1024 * 1024)).toFixed(1)} MB`
}

export default function OverviewPage() {
  const { projectId = '', snapshotId = '' } = useParams<{ projectId: string; snapshotId: string }>()
  const navigate = useNavigate()
  const ws = useWorkspace()
  const [caps, setCaps] = useState<CapabilityReport | null>(null)
  const [capsLoading, setCapsLoading] = useState(false)

  useEffect(() => {
    if (!projectId || !snapshotId) return
    setCapsLoading(true)
    getCapabilities(projectId, snapshotId)
      .then(r => setCaps(r))
      .catch(() => setCaps(null))
      .finally(() => setCapsLoading(false))
  }, [projectId, snapshotId])

  const files = ws.files
  const snap = ws.snapshot

  // Language breakdown
  const langCounts = files.reduce<Record<string, number>>((acc, f) => {
    const lang = f.language === 'unknown' ? 'other' : f.language
    acc[lang] = (acc[lang] ?? 0) + 1
    return acc
  }, {})
  const topLangs = Object.entries(langCounts)
    .sort((a, b) => b[1] - a[1])
    .slice(0, 6)

  const totalSize = files.reduce((n, f) => n + f.size, 0)

  // Suggested entry points: files with indicative names
  const ENTRY_NAMES = ['main.py', 'app.py', 'index.ts', 'index.js', 'main.ts', 'server.py',

  const entryPoints = files
    .filter(f => ENTRY_NAMES.some(name => f.path.endsWith(name) || f.name === name))
    .slice(0, 5)

  const sourceDisplayRef = snap?.source.resolved_commit
    ? `Commit ${snap.source.resolved_commit.slice(0, 12)}`
    : snap?.source.archive_digest
      ? `Archive ${snap.source.archive_digest.slice(0, 12)}`
      : 'Local archive'

  return (
    <div className="content-scroll">
      {/* ── Page header ── */}
      <div style={{ marginBottom: 24 }}>
        <p className="content-header__eyebrow" style={{ margin: '0 0 4px' }}>Overview</p>

          {ws.projectName ?? 'Repository'}
        </h1>
        {snap && (
          <p style={{ margin: '6px 0 0', fontSize: 12, color: 'var(--grepo-text-secondary)' }}>

          </p>
        )}

      {/* ── Stats ── */}

              <div className="skeleton skeleton--text" style={{ width: '60%' }} />
            </div>

        <ul className="stat-grid" aria-label="Repository statistics">
          <li className="stat-card">
            <span className="stat-card__label">Files</span>
            <span className="stat-card__value">{files.length.toLocaleString()}</span>
            <span className="stat-card__sub">inventoried</span>
          </li>
          <li className="stat-card">
            <span className="stat-card__label">Languages</span>
            <span className="stat-card__value">{topLangs.length}</span>
            <span className="stat-card__sub">detected</span>
          </li>
          <li className="stat-card">
            <span className="stat-card__label">Total size</span>

              <span className="stat-card__sub">syntax-analyzed</span>

import { useEffect, useState } from 'react'

import { SourceViewer } from '../../components/source/SourceViewer'
import { IntegrationPlaceholder } from '../../components/slots/IntegrationPlaceholder'
import { useWorkspace } from '../../contexts/WorkspaceContext'
import { getGraph, getAllFiles } from '../../services/v1/api'
import {
  buildExportManifest,
  downloadStructuralExport,
} from '../../services/exportService'
import type { Graph, GraphEntity } from '../../types/v1'

export default function MapPage() {
  const { projectId = '', snapshotId = '' } = useParams<{ projectId: string; snapshotId: string }>()
  const ws = useWorkspace()
  const [graph, setGraph] = useState<Graph | null>(ws.graph)
  const [graphLoading, setGraphLoading] = useState(false)
  const [graphError, setGraphError] = useState<string | null>(null)

  // Load graph
  useEffect(() => {
    if (!projectId || !snapshotId) return
    setGraphLoading(true)
    setGraphError(null)
    getGraph(projectId, snapshotId)
      .then(g => {
        setGraph(g)
        ws.setGraph(g)
      })
      .catch(err => setGraphError(err instanceof Error ? err.message : 'Failed to load graph.'))
      .finally(() => setGraphLoading(false))
  // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [projectId, snapshotId])

  function handleSelectEntity(entity: GraphEntity) {
    ws.selectEntity({
      entityId: entity.id,
      kind: entity.kind,
      path: entity.path ?? entity.label,
      fileId: entity.file_id,
    })
  }

  async function handleExport() {
    if (!graph || !ws.snapshot) return
    // Ensure we have file list for the manifest
    let fileList = ws.files
    if (fileList.length === 0) {
      fileList = await getAllFiles(projectId, snapshotId)
    }
    const filePage = { schema_version: '1.0' as const, snapshot_id: snapshotId, files: fileList, total: fileList.length, truncated: false }
    const manifest = buildExportManifest(ws.snapshot, graph, filePage)
    downloadStructuralExport(ws.snapshot, graph, manifest)
  }

  // Selected file for source panel
  const selectedFile = ws.selectedEntity?.fileId
    ? ws.files.find(f => f.id === ws.selectedEntity!.fileId)
    : null

  return (
    <div style={{ display: 'flex', flexDirection: 'column', height: '100%' }}>
      {/* ── Page header ── */}
      <div className="content-header">
        <p className="content-header__eyebrow">Structure</p>
        <h1 className="content-header__title">Repository Map</h1>
      </div>

      {/* ── Map + source panel ── */}
      <div style={{ flex: 1, display: 'grid', gridTemplateColumns: '1fr auto', minHeight: 0 }}>
        {/* Map */}
        <div style={{ minHeight: 0, overflow: 'hidden', position: 'relative' }}>
          {graphLoading && (
            <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'center', height: '100%', gap: 12 }}>
              <span className="spinner" aria-hidden="true" />
              <span style={{ color: 'var(--grepo-text-secondary)', fontSize: 13 }}>Building structural map…</span>
            </div>
          )}

          {graphError && (
            <div className="alert alert--error" style={{ margin: 24 }} role="alert">
              {graphError}
            </div>
          )}

          {graph && !graphLoading && (
            <StructureMap
              graph={graph}
              selectedEntityId={ws.selectedEntity?.entityId ?? null}
              onSelectEntity={handleSelectEntity}
              onExport={handleExport}
            />
          )}

          {!graph && !graphLoading && !graphError && (
            <div className="empty-state">
              <div className="empty-state__icon" aria-hidden="true">⊞</div>
              <h2 className="empty-state__title">No map data</h2>
              <p className="empty-state__body">
                Import a repository first, then open the Map.
              </p>
            </div>
          )}
        </div>

        {/* Source / context panel */}
        {selectedFile && (
          <div style={{ width: 420, borderLeft: '1px solid var(--grepo-line)', display: 'flex', flexDirection: 'column', overflow: 'hidden' }}>
            <div className="context-panel__header">
              <h2 className="context-panel__title">Source</h2>
            </div>
            <div style={{ flex: 1, overflow: 'hidden', padding: 12 }}>
              <SourceViewer
                projectId={projectId}
                snapshotId={snapshotId}
                fileId={selectedFile.id}
                path={selectedFile.path}
                language={selectedFile.language}
                lineStart={ws.selectedEntity?.lineRange?.start}
                lineEnd={ws.selectedEntity?.lineRange?.end}
              />
            </div>

            {/* INTEGRATION_SLOT: summaries.context-panel */}
            <div style={{ padding: '12px', borderTop: '1px solid var(--grepo-line)' }}>
              <IntegrationPlaceholder
                slotId="summaries.context-panel"
                title="AI Summaries"
                description="Integration slot: put the summaries feature here. Not connected yet."
                showDevDisclosure={false}
              />
            </div>
          </div>
        )}
      </div>

      {/* INTEGRATION_SLOT: dependencies.workspace */}
      {/* Dependencies overlay will be provided by the Dependencies feature module */}
    </div>
  )
}
